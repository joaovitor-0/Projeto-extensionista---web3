function selecionarInscrito(linha, id){ var linhas = document.querySelectorAll('.linhaInscrito'); linhas.forEach(function(item){ item.classList.remove('linhaSelecionada'); }); linha.classList.add('linhaSelecionada'); document.getElementById('inscritoSelecionado').value = id; } function adicionarInscrito(){ window.location.href = '/funcionarios/inscritos/adicionar/'; } function editarInscritoSelecionado(){ var id = document.getElementById('inscritoSelecionado').value; if(id == ''){ alert('Selecione um inscrito na tabela antes de editar.'); return; } window.location.href = '/funcionarios/inscritos/editar/' + id + '/'; } function excluirInscritoSelecionado(){ var id = document.getElementById('inscritoSelecionado').value; if(id == ''){ alert('Selecione um inscrito na tabela antes de excluir.'); return; } var confirmar = confirm('Tem certeza que deseja excluir o inscrito selecionado?'); if(confirmar){ window.location.href = '/funcionarios/inscritos/excluir/' + id + '/'; } } function confirmarAlteracaoInscrito(){ return confirm('Deseja salvar as alterações deste inscrito?'); } function ativarInscritoSelecionado() { let id = document.getElementById('inscritoSelecionado').value; if (id == '') { alert('Selecione um inscrito para ativar.'); } else { window.location.href = '/funcionarios/inscritos/ativar/' + id + '/'; } } function desativarInscritoSelecionado() { let id = document.getElementById('inscritoSelecionado').value; if (id == '') { alert('Selecione um inscrito para desativar.'); } else { window.location.href = '/funcionarios/inscritos/desativar/' + id + '/'; } }
